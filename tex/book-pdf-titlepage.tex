\usepackage{graphicx}

\definecolor{dategray}{HTML}{8A9098}

% Book-level release, matching `_quarto.yml` / `date:`.
% The updated stamp is the day this PDF is compiled (LuaTeX).
\newcommand{\bookreleasedate}{August 2, 2026}
\newcommand{\bookupdatedate}{%
  \directlua{
    local t = os.date("*t")
    local months = {
      "January","February","March","April","May","June",
      "July","August","September","October","November","December"
    }
    tex.sprint(months[t.month] .. " " .. tostring(t.day) .. ", " .. tostring(t.year))
  }%
}

\newcommand{\bookdatestamp}[2]{%
  {\sffamily\fontsize{7.4}{9}\selectfont\color{dategray}%
   \MakeUppercase{#1}\par}%
  \vspace{0.28em}%
  {\rmfamily\color{ink}#2\par}%
}

\newcommand{\bookreleasedates}{%
  \begin{minipage}{0.62\textwidth}
    \centering
    \begin{minipage}[t]{0.46\textwidth}
      \centering
      \bookdatestamp{Released}{\bookreleasedate}
    \end{minipage}%
    \hspace{1.4em}%
    \begin{minipage}[t]{0.46\textwidth}
      \centering
      \bookdatestamp{Last updated}{\bookupdatedate}
    \end{minipage}
  \end{minipage}%
}
